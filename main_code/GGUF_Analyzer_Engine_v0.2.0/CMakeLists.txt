cmake_minimum_required(VERSION 3.20)
project(GGUFAnalyzerEngine VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(GGUF_ENABLE_CUDA "Build the optional CUDA byte-statistics backend when a CUDA compiler is available" ON)
set(GGUF_SOURCES src/main.cpp src/gguf_engine.cpp src/backend_cpu.cpp)
set(GGUF_CUDA_ENABLED OFF)
if(GGUF_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    set(GGUF_CUDA_ENABLED ON)
    list(APPEND GGUF_SOURCES src/backend_cuda.cu)
  else()
    message(STATUS "CUDA compiler not found; building CPU backend only")
  endif()
endif()
add_executable(gguf_analyzer ${GGUF_SOURCES})
target_include_directories(gguf_analyzer PRIVATE src)
if(GGUF_CUDA_ENABLED)
  target_compile_definitions(gguf_analyzer PRIVATE GGUF_HAS_CUDA=1)
  set_target_properties(gguf_analyzer PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_SEPARABLE_COMPILATION ON)
endif()
if(MSVC)
  target_compile_options(gguf_analyzer PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/permissive-> $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
else()
  target_compile_options(gguf_analyzer PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall> $<$<COMPILE_LANGUAGE:CXX>:-Wextra> $<$<COMPILE_LANGUAGE:CXX>:-Wpedantic> $<$<COMPILE_LANGUAGE:CXX>:-Wconversion>)
endif()
message(STATUS "GGUF CUDA backend enabled: ${GGUF_CUDA_ENABLED}")
